rows=[]
def copy(original,destination):
    original=Path(original); destination=Path(destination)
    destination.parent.mkdir(parents=True,exist_ok=True)
    if destination.exists():
        raise FileExistsError(destination)
    source=record(original)
    shutil.copyfile(original,destination)
    saved=record(destination)
    assert (saved['sha256'],saved['bytes'])==(source['sha256'],source['bytes'])
    rows.append(dict(original=source,published=saved))
